cmake_minimum_required(VERSION 3.0.2)
project(fwis_v3_planner)
add_compile_options(-std=c++17 -O2 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp nav_msgs geometry_msgs std_msgs visualization_msgs fwis_common fwis_v3_msgs tf)
catkin_package(CATKIN_DEPENDS roscpp nav_msgs geometry_msgs std_msgs visualization_msgs fwis_common fwis_v3_msgs tf)
include_directories(${catkin_INCLUDE_DIRS})
add_executable(fwis_v3_planner_node src/planner_node.cpp)
add_dependencies(fwis_v3_planner_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(fwis_v3_planner_node ${catkin_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_planner_policy_history tests/test_policy_history.cpp)
  if(TARGET test_planner_policy_history)
    add_dependencies(test_planner_policy_history ${catkin_EXPORTED_TARGETS})
    target_link_libraries(test_planner_policy_history ${catkin_LIBRARIES})
  endif()
endif()
