cmake_minimum_required(VERSION 3.16)
project(fwis_hybrid_joint LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(fwis_planning_core REQUIRED)
# Standard CMake discovery first; a local unpacked CasADi SDK is also supported.
find_package(casadi CONFIG REQUIRED HINTS
  "$ENV{COPTRANS_CASADI_ROOT}/cmake"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.coptrans-python/casadi/cmake")
include(CTest)
set(COPTRANS_HYBRID_SANITIZERS "" CACHE STRING "Comma-separated compiler sanitizers (address,undefined or thread)")
if(COPTRANS_HYBRID_SANITIZERS)
  add_compile_options(-fsanitize=${COPTRANS_HYBRID_SANITIZERS} -fno-omit-frame-pointer)
  add_link_options(-fsanitize=${COPTRANS_HYBRID_SANITIZERS})
endif()
add_library(hybrid_casadi_object OBJECT src/casadi_solver.cpp)
set_target_properties(hybrid_casadi_object PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(hybrid_casadi_object PRIVATE cxx_std_17)
target_link_libraries(hybrid_casadi_object PRIVATE casadi::casadi)
add_library(fwis_hybrid_joint SHARED src/model.cpp src/search.cpp src/certificate.cpp $<TARGET_OBJECTS:hybrid_casadi_object>)
add_library(fwis_hybrid_joint::fwis_hybrid_joint ALIAS fwis_hybrid_joint)
target_compile_features(fwis_hybrid_joint PUBLIC cxx_std_17)
target_include_directories(fwis_hybrid_joint PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
# Do not propagate the SDK-specific C++ string ABI to ROS, evaluators or consumers.
target_link_libraries(fwis_hybrid_joint PUBLIC fwis_planning_core::fwis_planning_core PRIVATE $<TARGET_FILE:casadi::casadi>)
set_target_properties(fwis_hybrid_joint PROPERTIES INSTALL_RPATH_USE_LINK_PATH ON)
target_compile_options(fwis_hybrid_joint PRIVATE -Wall -Wextra -Wpedantic)
install(DIRECTORY include/ DESTINATION include)
install(TARGETS fwis_hybrid_joint EXPORT export_fwis_hybrid_joint LIBRARY DESTINATION lib)
find_package(jsoncpp REQUIRED)
add_executable(hybrid_joint_case benchmark/run_case.cpp)
target_link_libraries(hybrid_joint_case PRIVATE fwis_hybrid_joint JsonCpp::JsonCpp)
install(TARGETS hybrid_joint_case DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_executable(hybrid_solver_test test/solver_test.cpp)
  target_link_libraries(hybrid_solver_test PRIVATE fwis_hybrid_joint GTest::gtest_main)
  add_test(NAME hybrid_solver_test COMMAND hybrid_solver_test)
  set_tests_properties(hybrid_solver_test PROPERTIES ENVIRONMENT "OPENBLAS_NUM_THREADS=1;OMP_NUM_THREADS=1")
  # Link the same Ipopt plugin at load time for sanitizer tests. Dynamic
  # RTLD_DEEPBIND used by upstream CasADi conflicts with sanitizer runtimes.
  find_library(FWIS_IPOPT_PLUGIN casadi_nlpsol_ipopt HINTS
    "$ENV{COPTRANS_CASADI_ROOT}" "${casadi_DIR}/..")
  if(FWIS_IPOPT_PLUGIN)
    target_link_libraries(hybrid_solver_test PRIVATE "${FWIS_IPOPT_PLUGIN}")
    target_compile_definitions(hybrid_solver_test PRIVATE COPTRANS_TEST_LINKED_IPOPT=1)
    add_executable(hybrid_solver_smoke test/solver_sanitizer.cpp)
    target_link_libraries(hybrid_solver_smoke PRIVATE fwis_hybrid_joint "${FWIS_IPOPT_PLUGIN}")
    add_test(NAME hybrid_solver_smoke COMMAND hybrid_solver_smoke)
    set_tests_properties(hybrid_solver_smoke PROPERTIES ENVIRONMENT "OPENBLAS_NUM_THREADS=1;OMP_NUM_THREADS=1")
  endif()
  foreach(oracle IN ITEMS model derivative)
    add_executable(hybrid_${oracle}_oracle test/${oracle}_oracle.cpp)
    target_link_libraries(hybrid_${oracle}_oracle PRIVATE fwis_hybrid_joint)
    add_test(NAME hybrid_${oracle}_oracle COMMAND hybrid_${oracle}_oracle)
  endforeach()

  add_executable(hybrid_certificate_tests test/certificate_tests.cpp)
  target_link_libraries(hybrid_certificate_tests PRIVATE fwis_hybrid_joint GTest::gtest_main)
  add_test(NAME hybrid_certificate_tests COMMAND hybrid_certificate_tests)
  add_executable(hybrid_collision_oracle test/collision_oracle.cpp)
  target_link_libraries(hybrid_collision_oracle PRIVATE fwis_hybrid_joint)
  add_test(NAME hybrid_collision_oracle COMMAND hybrid_collision_oracle)
endif()
ament_export_dependencies(fwis_planning_core)
ament_export_targets(export_fwis_hybrid_joint HAS_LIBRARY_TARGET)
ament_package()
